Задание №4
#include <iostream>
#include <vector>
#include <set>
 int main(){
     
     std::vector<int> a = {3, 7, 1, 16, 4};
     std::vector<int> b = {17, 6, 2, 11, 47};
     
     std::set<int> s;
     
     for (int i=0; i < a.size();i++){
         s.insert(a[i]);
     }
     for (int i=0; i<b.size();i++){
         s.insert(b[i]);
     }
     
     for (int num : s){
         std::cout << num << "\n";
     }
     return 0;
 }

В программу алгоритма включаются библиотеки: ввода и вывода, редактируемых списков, множеств. Создаются два разных независимых списка. Создаётся множество. Запускается условный цикл с постоянно увеличивающейся переменной, отвечающую за место в списке, которое добавляется в множество (упорядоченный по возрастанию список). Такой алгоритм делается для обоих списков, чтобы числа из обоих списков находились в одном упорядоченном контейнере. Далее выписывается(выводится на экран) каждый элемент числа по порядку из обоих списков. Return 0 возвращает конечное значение.


Задание №5
#include <iostream>
#include <vector>
 int main(){
     
     std::vector<int> a = {1, 2, 3, 4, 6};
     int x;
     
     for (int i=a.size()-1; i>=0;i--){
         if(a[i] - a[i-1]==2){
             x=a[i]-1;
         }
     }
     
     
     
     
    std::cout << x;
     
     return 0;
 }
В программу алгоритма включаются библиотеки: ввода и вывода и редактируемых списков. Список изначально упорядочен по условию задачи. Задаём переменную икс, которую мы должны вычислить. Задаем конечный цикл с постоянно увеличивающееся переменной. Ищем индекс двух рядом стоящих массивов, между которыми разница будет 2 (>1). Берётся впереди стоящая переменная при совпадающем условии и вычитается единица, чтобы получить недостающее число. В конце выводится переменная. Return 0 возвращает конечное значение.

#include <iostream>
using namespace std;

int main() {
    int a, b;
    
    std::cin >> a >> b;
    
    
    a=a+b;
    
    b=a-b;

}
Включаем библиотеку ввода и вывода и используем « using namespace std» чтобы сэкономить время на написании и не ссылаться на библиотеку каждый раз припиской std::«…». Задаём два числа для алгоритма. Отправляем команду ввода для этих двух чисел чтобы пользователь мог задать числовое значение переменных. Параметр a становится суммой двух переменных. В параметре b из суммы переменных(a) вычисляется b. В параметре b получаем изначальное значение a. Из a(суммы двух переменных) вычитается параметр b (имеющий значение а теперь) и получаем в параметре а изначальное значение параметра b.
